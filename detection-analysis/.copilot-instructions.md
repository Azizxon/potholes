### Common Patterns
```python
# Example detection pipeline
class PotholeDetector:
    def __init__(self, model_config: ModelConfig):
        self.model = load_model(model_config.model_path)
        self.preprocessor = ImagePreprocessor(model_config.input_size)
        self.postprocessor = DetectionPostprocessor(model_config.confidence_threshold)
    
    async def detect_potholes(self, image_data: ImageData) -> List[PotholeDetection]:
        """Detect potholes in the provided image data."""
        try:
            # Preprocess image
            processed_image = self.preprocessor.process(image_data.image)
            
            # Run inference
            predictions = await self.model.predict(processed_image)
            
            # Post-process results
            detections = self.postprocessor.process(predictions, image_data.metadata)
            
            # Filter by confidence
            filtered_detections = [d for d in detections if d.confidence > 0.7]
            
            return filtered_detections
            
        except Exception as e:
            logger.error(f"Detection failed: {e}")
            raise
```
