# Detection & Analysis Module - Copilot Instructions

## Module Purpose
This module processes collected data to detect and analyze potholes using computer vision, machine learning, and signal processing techniques implemented with C# .NET 9 and ML.NET framework.

## Technology Stack
- **Framework**: C# with .NET 9
- **Machine Learning**: ML.NET and ONNX.NET
- **Image Processing**: ImageSharp or OpenCV.NET
- **Computer Vision**: Microsoft Cognitive Services SDK or OpenCV.NET
- **Model Inference**: ONNX Runtime for .NET
- **Logging**: Microsoft.Extensions.Logging
- **Configuration**: Microsoft.Extensions.Configuration

## Key Responsibilities
- Image analysis for pothole detection
- Machine learning model inference using ML.NET
- Severity assessment and classification
- Confidence scoring and filtering
- Feature extraction and measurement
- ONNX model integration for advanced AI models

```csharp
// Example detection pipeline service
public interface IPotholeDetectionService
{
    Task<Result<IEnumerable<PotholeDetection>>> DetectPotholesAsync(ImageData imageData, CancellationToken cancellationToken = default);
    Task<Result<PotholeClassification>> ClassifyPotholeAsync(PotholeDetection detection, CancellationToken cancellationToken = default);
}

public class PotholeDetectionService : IPotholeDetectionService
{
    private readonly MLContext _mlContext;
    private readonly ITransformer _detectionModel;
    private readonly IImageProcessor _imageProcessor;
    private readonly IDetectionPostProcessor _postProcessor;
    private readonly ILogger<PotholeDetectionService> _logger;
    private readonly DetectionConfiguration _config;

    public PotholeDetectionService(
        ITransformer detectionModel,
        IImageProcessor imageProcessor,
        IDetectionPostProcessor postProcessor,
        ILogger<PotholeDetectionService> logger,
        IOptions<DetectionConfiguration> config)
    {
        _mlContext = new MLContext(seed: 0);
        _detectionModel = detectionModel;
        _imageProcessor = imageProcessor;
        _postProcessor = postProcessor;
        _logger = logger;
        _config = config.Value;
    }

    public async Task<Result<IEnumerable<PotholeDetection>>> DetectPotholesAsync(
        ImageData imageData, 
        CancellationToken cancellationToken = default)
    {
        try
        {
            _logger.LogInformation("Starting pothole detection for image {ImageId}", imageData.Id);

            // Preprocess image
            var processedImage = await _imageProcessor.ProcessAsync(imageData, cancellationToken);
            if (!processedImage.IsSuccess)
            {
                return Result.Failure<IEnumerable<PotholeDetection>>(processedImage.Error);
            }

            // Create prediction engine
            var predictionEngine = _mlContext.Model.CreatePredictionEngine<ImageInput, DetectionOutput>(_detectionModel);
            
            // Run inference
            var input = new ImageInput { ImageData = processedImage.Value.Data };
            var prediction = predictionEngine.Predict(input);

            // Post-process results
            var detections = await _postProcessor.ProcessAsync(prediction, imageData.Metadata, cancellationToken);
            
            // Filter by confidence threshold
            var filteredDetections = detections.Where(d => d.Confidence > _config.ConfidenceThreshold);

            _logger.LogInformation("Detection completed. Found {Count} potholes", filteredDetections.Count());
            return Result.Success(filteredDetections);
        }
        catch (OperationCanceledException)
        {
            _logger.LogWarning("Pothole detection was cancelled");
            throw;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Pothole detection failed for image {ImageId}", imageData.Id);
            return Result.Failure<IEnumerable<PotholeDetection>>($"Detection failed: {ex.Message}");
        }
    }

    public async Task<Result<PotholeClassification>> ClassifyPotholeAsync(
        PotholeDetection detection, 
        CancellationToken cancellationToken = default)
    {
        try
        {
            // Classification logic using ML.NET
            var classificationResult = await PerformClassificationAsync(detection, cancellationToken);
            return Result.Success(classificationResult);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Classification failed for detection {DetectionId}", detection.Id);
            return Result.Failure<PotholeClassification>($"Classification failed: {ex.Message}");
        }
    }

    private async Task<PotholeClassification> PerformClassificationAsync(
        PotholeDetection detection, 
        CancellationToken cancellationToken)
    {
        // Implementation for severity and type classification
        await Task.Delay(1, cancellationToken); // Simulate async work
        
        return new PotholeClassification
        {
            Severity = CalculateSeverity(detection),
            Type = DetermineType(detection),
            Confidence = detection.Confidence
        };
    }
}
```

### ONNX Model Integration
```csharp
// Service for ONNX model inference
public class OnnxInferenceService : IOnnxInferenceService
{
    private readonly InferenceSession _session;
    private readonly ILogger<OnnxInferenceService> _logger;

    public OnnxInferenceService(string modelPath, ILogger<OnnxInferenceService> logger)
    {
        var sessionOptions = new SessionOptions();
        sessionOptions.EnableCpuMemArena = false;
        sessionOptions.EnableMemoryPattern = false;
        
        _session = new InferenceSession(modelPath, sessionOptions);
        _logger = logger;
    }

    public async Task<float[]> RunInferenceAsync(float[] inputData)
    {
        try
        {
            var inputMeta = _session.InputMetadata;
            var inputName = inputMeta.Keys.First();
            
            var tensor = new DenseTensor<float>(inputData, new[] { 1, inputData.Length });
            var inputs = new List<NamedOnnxValue> { NamedOnnxValue.CreateFromTensor(inputName, tensor) };
            
            using var results = _session.Run(inputs);
            var output = results.First().AsEnumerable<float>().ToArray();
            
            return output;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "ONNX inference failed");
            throw;
        }
    }

    public void Dispose()
    {
        _session?.Dispose();
    }
}
```
