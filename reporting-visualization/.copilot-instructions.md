# Reporting & Visualization Module - Copilot Instructions

## Module Purpose
This module provides user interfaces, reports, and visualizations for pothole data. It includes web dashboards, mobile interfaces, and automated reporting capabilities using Angular frontend with .NET backend.

- **UI Components**: Angular Material or PrimeNG
- **Maps**: Leaflet or Google Maps API
- **Charts**: Chart.js or D3.js
- **State Management**: NgRx (for complex state)
- **HTTP Client**: Angular HttpClient

## Key Responsibilities
- Angular web dashboard development

#### .NET API Controller Example
```csharp
[ApiController]
[Route("api/v1/[controller]")]
public class PotholesController : ControllerBase
{
    private readonly IPotholeService _potholeService;
    private readonly IMapper _mapper;
    private readonly ILogger<PotholesController> _logger;

    public PotholesController(
        IPotholeService potholeService,
        IMapper mapper,
        ILogger<PotholesController> logger)
    {
        _potholeService = potholeService;
        _mapper = mapper;
        _logger = logger;
    }

    [HttpGet]
    public async Task<ActionResult<PagedResult<PotholeDto>>> GetPotholes(
        [FromQuery] double latMin,
        [FromQuery] double latMax,
        [FromQuery] double lonMin,
        [FromQuery] double lonMax,
        [FromQuery] PotholeSeverity? severity = null,
        [FromQuery] int page = 1,
        [FromQuery] int pageSize = 100)
    {
        try
        {
            var query = new PotholeQuery
            {
                LatMin = latMin,
                LatMax = latMax,
                LonMin = lonMin,
                LonMax = lonMax,
                Severity = severity,
                Page = page,
                PageSize = Math.Min(pageSize, 1000)
            };

            var result = await _potholeService.GetPotholesAsync(query);
            if (!result.IsSuccess)
            {
                return BadRequest(result.Error);
            }

            var dto = _mapper.Map<PagedResult<PotholeDto>>(result.Value);
            return Ok(dto);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to retrieve potholes");
            return StatusCode(500, "Internal server error");
        }
    }

    [HttpPost]
    public async Task<ActionResult<string>> CreatePothole([FromBody] CreatePotholeRequest request)
    {
        try
        {
            var pothole = _mapper.Map<PotholeDetection>(request);
            var result = await _potholeService.CreatePotholeAsync(pothole);
            
            if (!result.IsSuccess)
            {
                return BadRequest(result.Error);
            }

            return CreatedAtAction(nameof(GetPothole), new { id = result.Value }, result.Value);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to create pothole");
            return StatusCode(500, "Internal server error");
        }
    }
}
```

#### Angular Service Example
```typescript
@Injectable({
  providedIn: 'root'
})
export class PotholeService {
  private readonly apiUrl = `${environment.apiBaseUrl}/api/v1/potholes`;

  constructor(
    private http: HttpClient,
    private logger: Logger
  ) {}

  getPotholes(query: PotholeQuery): Observable<PagedResult<Pothole>> {
    const params = new HttpParams()
      .set('latMin', query.latMin.toString())
      .set('latMax', query.latMax.toString())
      .set('lonMin', query.lonMin.toString())
      .set('lonMax', query.lonMax.toString())
      .set('page', query.page.toString())
      .set('pageSize', query.pageSize.toString());

    if (query.severity) {
      params.set('severity', query.severity);
    }

    return this.http.get<PagedResult<Pothole>>(this.apiUrl, { params })
      .pipe(
        catchError(this.handleError<PagedResult<Pothole>>('getPotholes'))
      );
  }

  createPothole(pothole: CreatePotholeRequest): Observable<string> {
    return this.http.post<string>(this.apiUrl, pothole)
      .pipe(
        catchError(this.handleError<string>('createPothole'))
      );
  }

  private handleError<T>(operation = 'operation') {
    return (error: HttpErrorResponse): Observable<T> => {
      this.logger.error(`${operation} failed: ${error.message}`);
      return throwError(() => error);
    };
  }
}
```

#### Angular Component Example
```typescript
@Component({
  selector: 'app-pothole-map',
  templateUrl: './pothole-map.component.html',
  styleUrls: ['./pothole-map.component.scss']
})
export class PotholeMapComponent implements OnInit, OnDestroy {
  private map?: L.Map;
  private markers: L.MarkerClusterGroup = L.markerClusterGroup();
  private destroy$ = new Subject<void>();

  potholes$ = this.potholeService.potholes$;
  loading$ = this.potholeService.loading$;
  error$ = this.potholeService.error$;

  constructor(
    private potholeService: PotholeService,
    private cdr: ChangeDetectorRef
  ) {}

  ngOnInit(): void {
    this.initializeMap();
    this.subscribeToData();
  }

  ngOnDestroy(): void {
    this.destroy$.next();
    this.destroy$.complete();
    this.map?.remove();
  }

  private initializeMap(): void {
    this.map = L.map('map').setView([40.7128, -74.0060], 13);
    
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '© OpenStreetMap contributors'
    }).addTo(this.map);

    this.map.addLayer(this.markers);
  }

  private subscribeToData(): void {
    this.potholes$
      .pipe(takeUntil(this.destroy$))
      .subscribe(potholes => {
        this.updateMarkers(potholes);
      });
  }

  private updateMarkers(potholes: Pothole[]): void {
    this.markers.clearLayers();
    
    potholes.forEach(pothole => {
      const marker = L.marker([pothole.latitude, pothole.longitude])
        .bindPopup(this.createPopupContent(pothole));
      
      this.markers.addLayer(marker);
    });
  }

  private createPopupContent(pothole: Pothole): string {
    return `
      <div class="pothole-popup">
        <h4>Pothole ${pothole.id}</h4>
        <p><strong>Severity:</strong> ${pothole.severity}</p>
        <p><strong>Confidence:</strong> ${(pothole.confidence * 100).toFixed(1)}%</p>
        <p><strong>Detected:</strong> ${new Date(pothole.detectedAt).toLocaleDateString()}</p>
      </div>
    `;
  }
}
```

#### SignalR Hub for Real-time Updates
```csharp
[Authorize]
public class PotholeHub : Hub
{
    private readonly ILogger<PotholeHub> _logger;

    public PotholeHub(ILogger<PotholeHub> logger)
    {
        _logger = logger;
    }

    public async Task JoinLocationGroup(double latitude, double longitude, double radius)
    {
        var groupName = $"location_{latitude:F4}_{longitude:F4}_{radius}";
        await Groups.AddToGroupAsync(Context.ConnectionId, groupName);
        _logger.LogInformation("User {UserId} joined location group {GroupName}", 
            Context.UserIdentifier, groupName);
    }

    public async Task LeaveLocationGroup(double latitude, double longitude, double radius)
    {
        var groupName = $"location_{latitude:F4}_{longitude:F4}_{radius}";
        await Groups.RemoveFromGroupAsync(Context.ConnectionId, groupName);
    }
}

// Service to send real-time updates
public interface IPotholeNotificationService
{
    Task NotifyPotholeDetectedAsync(PotholeDocument pothole);
    Task NotifyPotholeStatusChangedAsync(string potholeId, PotholeStatus newStatus);
}

public class PotholeNotificationService : IPotholeNotificationService
{
    private readonly IHubContext<PotholeHub> _hubContext;
    private readonly ILogger<PotholeNotificationService> _logger;

    public PotholeNotificationService(
        IHubContext<PotholeHub> hubContext,
        ILogger<PotholeNotificationService> logger)
    {
        _hubContext = hubContext;
        _logger = logger;
    }

    public async Task NotifyPotholeDetectedAsync(PotholeDocument pothole)
    {
        var notification = new
        {
            Type = "PotholeDetected",
            Data = pothole
        };

        await _hubContext.Clients.All.SendAsync("PotholeUpdate", notification);
        _logger.LogInformation("Sent pothole detection notification for {PotholeId}", pothole.Id);
    }
}
```
