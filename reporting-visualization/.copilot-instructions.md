```python
# Example API endpoint
from fastapi import FastAPI, HTTPException, Query
from typing import List, Optional

app = FastAPI()

@app.get("/api/v1/potholes")
async def get_potholes(
    lat_min: float = Query(..., ge=-90, le=90),
    lat_max: float = Query(..., ge=-90, le=90),
    lon_min: float = Query(..., ge=-180, le=180),
    lon_max: float = Query(..., ge=-180, le=180),
    severity: Optional[str] = Query(None, regex="^(low|medium|high|critical)$"),
    limit: int = Query(100, ge=1, le=1000)
) -> List[PotholeResponse]:
    """Get potholes within a bounding box."""
    try:
        # Validate bounding box
        if lat_min >= lat_max or lon_min >= lon_max:
            raise HTTPException(400, "Invalid bounding box")
        
        # Query database
        potholes = await pothole_service.get_potholes_in_bounds(
            lat_min, lat_max, lon_min, lon_max, severity, limit
        )
        
        # Transform to response format
        return [PotholeResponse.from_entity(p) for p in potholes]
        
    except Exception as e:
        logger.error(f"Failed to get potholes: {e}")
        raise HTTPException(500, "Internal server error")

# Example visualization component
class PotholeMapVisualization:
    def __init__(self, map_config: MapConfig):
        self.config = map_config
    
    def generate_heatmap(self, potholes: List[Pothole]) -> HeatmapData:
        """Generate heatmap data for pothole visualization."""
        return {
            'points': [
                {
                    'lat': p.latitude,
                    'lng': p.longitude,
                    'intensity': self._calculate_intensity(p)
                }
                for p in potholes
            ],
            'config': self.config.heatmap_settings
        }
```
