# Database & Storage Module - Copilot Instructions

## Module Purpose
This module manages persistent storage of pothole data, detection results, metadata, and analytics using MongoDB. It provides data access layers and ensures data consistency across the application.

## Technology Stack
- **Framework**: C# with .NET 9
- **Database**: MongoDB with spatial indexing
- **ODM**: MongoDB.Driver and MongoDB.Entities
- **Caching**: Microsoft.Extensions.Caching (Redis optional)
- **Migration**: Custom migration scripts for MongoDB

## Key Responsibilities
- MongoDB database schema design and management
- Data persistence and retrieval operations
- Spatial indexing for GPS-based queries (GeoJSON)
- Data backup and archival strategies
- Performance optimization for large datasets

```csharp
// Example MongoDB document model
[BsonCollection("potholes")]
public class PotholeDocument : IDocument
{
    [BsonId]
    [BsonRepresentation(BsonType.ObjectId)]
    public string Id { get; set; } = string.Empty;

    [BsonElement("location")]
    public GeoJsonPoint<GeoJson2DGeographicCoordinates> Location { get; set; } = null!;

    [BsonElement("severity")]
    [BsonRepresentation(BsonType.String)]
    public PotholeSeverity Severity { get; set; }

    [BsonElement("confidence")]
    public double Confidence { get; set; }

    [BsonElement("imagePath")]
    public string ImagePath { get; set; } = string.Empty;

    [BsonElement("detectedAt")]
    public DateTime DetectedAt { get; set; }

    [BsonElement("reportedBy")]
    public string? ReportedBy { get; set; }

    [BsonElement("status")]
    [BsonRepresentation(BsonType.String)]
    public PotholeStatus Status { get; set; } = PotholeStatus.Detected;

    [BsonElement("metadata")]
    public Dictionary<string, object>? Metadata { get; set; }
}

// Example repository pattern
public interface IPotholeRepository
{
    Task<Result<string>> SaveDetectionAsync(PotholeDetection detection);
    Task<Result<PotholeDocument?>> GetByIdAsync(string id);
    Task<Result<List<PotholeDocument>>> GetByLocationAsync(double latitude, double longitude, double radiusKm);
    Task<Result<bool>> UpdateStatusAsync(string id, PotholeStatus status);
}

public class PotholeRepository : IPotholeRepository
{
    private readonly IMongoCollection<PotholeDocument> _collection;
    private readonly ILogger<PotholeRepository> _logger;

    public PotholeRepository(IMongoDatabase database, ILogger<PotholeRepository> logger)
    {
        _collection = database.GetCollection<PotholeDocument>("potholes");
        _logger = logger;
    }

    public async Task<Result<string>> SaveDetectionAsync(PotholeDetection detection)
    {
        try
        {
            var document = new PotholeDocument
            {
                Location = new GeoJsonPoint<GeoJson2DGeographicCoordinates>(
                    new GeoJson2DGeographicCoordinates(detection.Location.Longitude, detection.Location.Latitude)),
                Severity = detection.Severity,
                Confidence = detection.Confidence,
                ImagePath = detection.ImagePath,
                DetectedAt = DateTime.UtcNow,
                ReportedBy = detection.ReportedBy,
                Metadata = detection.Metadata
            };

            await _collection.InsertOneAsync(document);
            
            _logger.LogInformation("Pothole detection saved with ID: {Id}", document.Id);
            return Result.Success(document.Id);
        }
        catch (MongoException ex)
        {
            _logger.LogError(ex, "Failed to save pothole detection");
            return Result.Failure<string>($"Database error: {ex.Message}");
        }
    }

    public async Task<Result<List<PotholeDocument>>> GetByLocationAsync(double latitude, double longitude, double radiusKm)
    {
        try
        {
            var point = new GeoJsonPoint<GeoJson2DGeographicCoordinates>(
                new GeoJson2DGeographicCoordinates(longitude, latitude));

            var filter = Builders<PotholeDocument>.Filter.Near(
                x => x.Location, 
                point, 
                radiusKm * 1000); // Convert to meters

            var potholes = await _collection
                .Find(filter)
                .Sort(Builders<PotholeDocument>.Sort.Ascending(x => x.DetectedAt))
                .ToListAsync();

            return Result.Success(potholes);
        }
        catch (MongoException ex)
        {
            _logger.LogError(ex, "Failed to query potholes by location");
            return Result.Failure<List<PotholeDocument>>($"Query failed: {ex.Message}");
        }
    }
    }
}

// Service layer example
public interface IPotholeService
{
    Task<Result<string>> CreatePotholeAsync(PotholeDetection detection);
    Task<Result<PagedResult<PotholeDocument>>> GetPotholesAsync(PotholeQuery query);
    Task<Result<List<PotholeDocument>>> GetNearbyPotholesAsync(double latitude, double longitude, double radiusKm);
}

public class PotholeService : IPotholeService
{
    private readonly IPotholeRepository _repository;
    private readonly IMemoryCache _cache;
    private readonly ILogger<PotholeService> _logger;

    public PotholeService(
        IPotholeRepository repository,
        IMemoryCache cache,
        ILogger<PotholeService> logger)
    {
        _repository = repository;
        _cache = cache;
        _logger = logger;
    }

    public async Task<Result<List<PotholeDocument>>> GetNearbyPotholesAsync(double latitude, double longitude, double radiusKm)
    {
        var cacheKey = $"nearby_potholes_{latitude}_{longitude}_{radiusKm}";
        
        if (_cache.TryGetValue(cacheKey, out List<PotholeDocument>? cachedResult))
        {
            return Result<List<PotholeDocument>>.Success(cachedResult!);
        }

        var result = await _repository.GetByLocationAsync(latitude, longitude, radiusKm);
        if (result.IsSuccess)
        {
            _cache.Set(cacheKey, result.Value, TimeSpan.FromMinutes(5));
        }

        return result;
    }
}

// MongoDB configuration example
public static class MongoDbConfiguration
{
    public static IServiceCollection AddMongoDb(this IServiceCollection services, IConfiguration configuration)
    {
        var connectionString = configuration.GetConnectionString("MongoDB");
        var databaseName = configuration["MongoDB:DatabaseName"];

        services.AddSingleton<IMongoClient>(_ => new MongoClient(connectionString));
        services.AddSingleton<IMongoDatabase>(serviceProvider =>
        {
            var client = serviceProvider.GetService<IMongoClient>();
            return client!.GetDatabase(databaseName);
        });

        // Register repositories
        services.AddScoped<IPotholeRepository, PotholeRepository>();
        services.AddScoped<IPotholeService, PotholeService>();

        return services;
    }

    public static async Task EnsureIndexesAsync(IMongoDatabase database)
    {
        var collection = database.GetCollection<PotholeDocument>("potholes");

        // Create 2dsphere index for geospatial queries
        var locationIndexModel = new CreateIndexModel<PotholeDocument>(
            Builders<PotholeDocument>.IndexKeys.Geo2DSphere(x => x.Location),
            new CreateIndexOptions { Name = "location_2dsphere" });

        // Create compound index for common queries
        var compoundIndexModel = new CreateIndexModel<PotholeDocument>(
            Builders<PotholeDocument>.IndexKeys
                .Ascending(x => x.Status)
                .Descending(x => x.DetectedAt),
            new CreateIndexOptions { Name = "status_detected_at" });

        await collection.Indexes.CreateManyAsync(new[] { locationIndexModel, compoundIndexModel });
    }
}
```

### Migration Strategy
```csharp
public class DatabaseMigration
{
    private readonly IMongoDatabase _database;
    private readonly ILogger<DatabaseMigration> _logger;

    public async Task MigrateAsync()
    {
        await EnsureCollectionsExistAsync();
        await MongoDbConfiguration.EnsureIndexesAsync(_database);
        await SeedInitialDataAsync();
    }

    private async Task EnsureCollectionsExistAsync()
    {
        var collections = await _database.ListCollectionNamesAsync();
        var existingCollections = await collections.ToListAsync();

        if (!existingCollections.Contains("potholes"))
        {
            await _database.CreateCollectionAsync("potholes");
        }
    }
}
```
