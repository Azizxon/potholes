### Common Patterns
```python
# Example repository pattern
class PotholeRepository:
    def __init__(self, db_session: Session):
        self.db = db_session
    
    async def save_detection(self, detection: PotholeDetection) -> str:
        """Save a pothole detection to the database."""
        try:
            # Create database entity
            db_pothole = Pothole(
                id=generate_id(),
                latitude=detection.location.latitude,
                longitude=detection.location.longitude,
                severity=detection.severity,
                confidence=detection.confidence,
                image_path=detection.image_path,
                detected_at=detection.timestamp,
                created_at=datetime.utcnow()
            )
            
            # Save to database
            self.db.add(db_pothole)
            await self.db.commit()
            
            return db_pothole.id
            
        except Exception as e:
            await self.db.rollback()
            logger.error(f"Failed to save detection: {e}")
            raise
    
    async def find_nearby_potholes(self, location: GPSCoordinate, radius_km: float) -> List[Pothole]:
        """Find potholes within a specified radius."""
        query = self.db.query(Pothole).filter(
            spatial_distance(Pothole.location, location) <= radius_km
        )
        return await query.all()
```
